# Wilderness-only copies of the three Ascension Worlds superproject planet classes.
#
# Localisation is not trigger-conditional, so a Wilderness empire cannot simply see a
# different name on pc_pd_agriworld. These copies keep every stat identical to PD's
# classes and only change the class ID, which gives them their own localised name.
# The decisions give the _wilderness class to Wilderness empires and the original class
# to everyone else; pd_aw_is_superproject_class is overridden to recognise both sets.
#
# Loaded after planetary_diversity_ascension_worlds.

# Core Tap
pc_pd_core_tap_wilderness = {
	entity = "pd_core_tap_planet"
	entity_scale = @planet_standard_scale
	icon = GFX_planet_type_core_tap
	icon_large = GFX_planet_type_core_tap_big
	#appearance
	show_city = no
	atmosphere_color 		= hsv { 0.48 0.15 0.8 }
	atmosphere_intensity 	= 1.0
	atmosphere_width 		= 0.5
	city_color_lut = "gfx/portraits/misc/colorcorrection_neutral.dds"
	#planet defines
	#ideal = yes
	colonizable = yes
	starting_planet = no
	#colonization_tech = { }
	planet_size = { min = @habitable_planet_min_size max = @habitable_planet_max_size }
	moon_size = { min = @habitable_moon_min_size max = @habitable_moon_max_size }
	district_set = standard
	carry_cap_per_free_district = @carry_cap_low
	#is_artificial_planet = yes
	modifier = {
		pop_environment_tolerance = 0.25
		biological_pop_happiness = -0.1
		lithoid_pop_happiness = -0.1
		habitability_ceil_add = -0.25
	}
	uses_alternative_skies_for_moons = no
	#is_artificial_planet = yes
}
