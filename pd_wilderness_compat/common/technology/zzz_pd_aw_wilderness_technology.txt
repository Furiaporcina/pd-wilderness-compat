# Wilderness compatibility override for the Planetary Diversity Ascension Worlds agriworld technology.
# Loaded after planetary_diversity_ascension_worlds, so this key replaces the original.
#
# Why this exists:
#   The original technology requires tech_nano_vitality_crops, a vanilla technology whose
#   potential is "is_wilderness_empire = no". A Wilderness empire can therefore never
#   research it, which silently locks decision_pd_aw_agriworld out of the game.
#
# What changes:
#   - tech_nano_vitality_crops is removed from the shared prerequisite list and moved into
#     the potential block as a non-Wilderness-only requirement. Non-Wilderness empires still
#     need it before the technology becomes available, so their behaviour is unchanged.
#   - Wilderness empires only need tech_food_processing_2 and tech_ecological_adaptation,
#     both of which vanilla lets them research (food processing 2 has a Wilderness swap).
#   - The "country_uses_food = no" weight penalty is limited to non-Wilderness empires,
#     because Wilderness economies spend biomass instead of food and would otherwise be
#     pushed to weight 0.
#   Every other field (cost, tier, category, tooltip, ai_weight) is copied verbatim.

tech_pd_planetary_agriworld = {
	area = society
	cost = @tier3cost3
	tier = 3
	category = { biology }
	prerequisites = { "tech_food_processing_2" "tech_ecological_adaptation" }
	weight = @tier3weight3
	is_rare = yes

	potential = {
		OR = {
			is_wilderness_empire = yes
			has_technology = tech_nano_vitality_crops
		}
	}

	modifier = {
		custom_tooltip = tech_pd_planetary_agriworld_tooltip
	}

	weight_modifier = {
		factor = 2
		modifier = {
			factor = 0
			AND = {
				is_wilderness_empire = no
				country_uses_food = no
			}
		}
		modifier = {
			factor = 0
			OR = {
				has_civic = civic_environmentalist
				has_civic = civic_agrarian_idyll
			}
		}
		modifier = {
			factor = 1.25
			has_tradition = tr_prosperity_adopt
		}
		modifier = {
			factor = value:tech_weight_likelihood
			has_trait_in_council = { TRAIT = leader_trait_expertise_biology }
		}
		modifier = {
			factor = 1.5
			is_specialist_subject_type = { TYPE = prospectorium }
		}
	}

	ai_weight = {
		factor = @farming_building_ai_weight_factor # food processing
		modifier = {
			factor = 1.25
			has_trait_in_council = { TRAIT = leader_trait_expertise_biology }
		}
	}
}
